const bcrypt = require('bcryptjs'); const sqlite3 = require('sqlite3').verbose(); const fs = require('fs'); const path = require('path'); const dbPath = path.join(__dirname, 'clinic_inventory.db'); const schemaPath = path.join(__dirname, 'schema.sql'); const seedPath = path.join(__dirname, 'seed-data.sql'); const db = new sqlite3.Database(dbPath); const hashPassword = async (password) => { const salt = await bcrypt.genSalt(10); return bcrypt.hash(password, salt); }; const runSqlFile = (filePath) => { const sql = fs.readFileSync(filePath, 'utf8'); return new Promise((resolve, reject) => { db.exec(sql, (err) => { if (err) reject(err); else resolve(); }); }); }; (async () => { try { await runSqlFile(schemaPath); const adminPassword = await hashPassword('admin123'); const staffPassword = await hashPassword('staff123'); db.run( `INSERT OR IGNORE INTO users (email, password_hash, role, full_name) VALUES (?, ?, ?, ?)`, ['admin@clinic.com', adminPassword, 'Admin', 'System Admin'] ); db.run( `INSERT OR IGNORE INTO users (email, password_hash, role, full_name) VALUES (?, ?, ?, ?)`, ['staff@clinic.com', staffPassword, 'Staff', 'Clinic Staff'] ); await runSqlFile(seedPath); console.log('Seed data completed successfully.'); db.close(); } catch (error) { console.error('Seed error:', error); db.close(); } })();
